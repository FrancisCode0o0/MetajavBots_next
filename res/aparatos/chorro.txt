/**
 * MOTORACHORRO «Jetzazo» — Un empujón de fuego: durante 2/3 de
 * segundo vas 2,6 veces más rápido.
 *
 * Cómo conectarlo a tu robot:
 *   1. Ve hasta él y tócalo: lo coges (aparatos() te dice dónde
 *      está).
 *   2. Conéctalo en su hueco (si no es ese, no engancha):
 *      conectar("espalda");
 *   3. Úsalo: usarAparato();
 * Se recarga en 4 s.
 *
 * Cumple la interfaz Aparato: tipo(), nombre(), conectado(),
 * listo()…
 */
public final class MotorAChorro implements Aparato {

    /** Los turnos que faltan para poder usarlo otra vez. */
    private int espera;

    @Override
    public String tipo() {
        return "chorro";
    }

    /** Lo que hace al usarlo. */
    void usar(Metajavbot robot) {
        if (espera > 0) return;
        robot.acelerar(20, 2.6);   // 20 turnos, 2,6 veces
        espera = 120;
    }
}

/**
 * DUESENTRIEBWERK «Jetzazo» — Ein Feuerstoß: 2/3 Sekunden lang
 * fährst du 2,6-mal so schnell.
 *
 * So schließt du es an deinen Roboter an:
 *   1. Fahr hin und berühre es: Du hebst es auf (geraete() sagt
 *      dir, wo es ist).
 *   2. Schließ es an seinem Platz an (ein anderer hält nicht):
 *      anschliessen("ruecken");
 *   3. Benutze es: geraetBenutzen();
 * Lädt in 4 s wieder auf.
 *
 * Erfüllt das Interface Geraet: typ(), name(), angeschlossen(),
 * bereit() …
 */
public final class Duesentriebwerk implements Geraet {

    /** Die Züge, bis man es wieder benutzen kann. */
    private int warten;

    @Override
    public String typ() {
        return "chorro";
    }

    /** Was es beim Benutzen tut. */
    void benutzen(MetajavbotDe roboter) {
        if (warten > 0) return;
        roboter.beschleunigen(20, 2.6);   // 20 Züge, 2,6-mal
        warten = 120;
    }
}
== en
/**
 * JETENGINE «Jetzazo» — A burst of fire: for 2/3 of a second you go
 * 2.6 times faster.
 *
 * How to plug it into your robot:
 *   1. Drive to it and touch it: you pick it up (gadgets() tells
 *      you where it is).
 *   2. Plug it into its slot (any other won't hold):
 *      plugIn("back");
 *   3. Use it: useGadget();
 * Recharges in 4 s.
 *
 * Implements the Gadget interface: type(), name(), pluggedIn(),
 * ready()…
 */
public final class JetEngine implements Gadget {

    /** The ticks left until it can be used again. */
    private int wait;

    @Override
    public String type() {
        return "chorro";
    }

    /** What it does when used. */
    void use(MetajavbotEn robot) {
        if (wait > 0) return;
        robot.boost(20, 2.6);   // 20 ticks, 2.6 times
        wait = 120;
    }
}
