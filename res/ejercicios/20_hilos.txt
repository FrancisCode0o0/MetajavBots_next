plantilla:

    for (int i = 0; i < datos.length / 2; i++) {
        suma += datos[i];
    }
    return suma;
solucion:
    int mitad = datos.length / 2;
    long[] parciales = new long[2];
    Thread primera = new Thread(() -> {
        for (int i = 0; i < mitad; i++) {
            parciales[0] += datos[i];
        }
    });
    Thread segunda = new Thread(() -> {
        for (int i = mitad; i < datos.length; i++) {
            parciales[1] += datos[i];
        }
    });
    primera.start();
    segunda.start();
    primera.join();
    segunda.join();
    return parciales[0] + parciales[1];

    import java.util.concurrent.ExecutorService;
    import java.util.concurrent.Executors;
    import java.util.concurrent.TimeUnit;
    import java.util.concurrent.atomic.AtomicInteger;

    public class Municion {

        public static int contarDisparos(int robots, int disparosPorRobot) throws InterruptedException {
            AtomicInteger cuenta = new AtomicInteger();
            ExecutorService pool = Executors.newFixedThreadPool(4);
            for (int r = 0; r < robots; r++) {
                pool.submit(() -> {
                    for (int i = 0; i < disparosPorRobot; i++) {
                        cuenta.incrementAndGet();
                    }
                });

            pool.shutdown();
            pool.awaitTermination(10, TimeUnit.SECONDS);
            return cuenta.get();
        }
    }

    import java.util.List;
    import java.util.Map;
    import java.util.concurrent.ConcurrentHashMap;
    import java.util.concurrent.ExecutorService;
    import java.util.concurrent.Executors;
    import java.util.concurrent.TimeUnit;

    public class Marcador {

        public static Map<String, Integer> danoPorRobot(List<String> golpes) throws InterruptedException {
            Map<String, Integer> marcador = new ConcurrentHashMap<>();
            ExecutorService pool = Executors.newFixedThreadPool(4);
            for (String golpe : golpes) {
                pool.submit(() -> {
                    String[] partes = golpe.split(":");
                    marcador.merge(partes[0], Integer.parseInt(partes[1]), Integer::sum);
                });

            pool.shutdown();
            pool.awaitTermination(10, TimeUnit.SECONDS);
            return marcador;
        }
    }

    import java.util.ArrayList;
    import java.util.List;
    import java.util.concurrent.Callable;
    import java.util.concurrent.ExecutorService;
    import java.util.concurrent.Executors;
    import java.util.concurrent.Future;

    public class Balistica {

        public static List<Integer> danos(List<Integer> distancias) throws Exception {
            List<Callable<Integer>> tareas = new ArrayList<>();
            for (int d : distancias) {
                tareas.add(() -> Math.max(0, 100 - d / 10));
            }
            ExecutorService pool = Executors.newFixedThreadPool(3);
            try {
                List<Integer> resultados = new ArrayList<>();
                for (Future<Integer> futuro : pool.invokeAll(tareas)) {
                    resultados.add(futuro.get());
                }
                return resultados;
            } finally {
                pool.shutdown();
            }
        }
    }

    import java.util.concurrent.CompletableFuture;

    public class Ataque {

        public static String combinado(int laser, int misil) {
            CompletableFuture<Integer> rayo = CompletableFuture.supplyAsync(() -> calcular("láser", laser));
            CompletableFuture<Integer> cohete = CompletableFuture.supplyAsync(() -> calcular("misil", misil));

                    .thenApply(total -> "total: " + total)
                    .exceptionally(e -> "fallo: " + e.getCause().getMessage())
                    .join();

    import java.util.ArrayList;
    import java.util.List;
    import java.util.Optional;
    import java.util.concurrent.ArrayBlockingQueue;
    import java.util.concurrent.BlockingQueue;

    public class Enlace {

        public static List<String> transmitir(List<String> mensajes) throws InterruptedException {
            BlockingQueue<Optional<String>> cola = new ArrayBlockingQueue<>(2);
            List<String> recibidos = new ArrayList<>();
            Thread emisor = new Thread(() -> {
                try {
                    for (String m : mensajes) {
                        cola.put(Optional.of(m));
                    }
                    cola.put(Optional.empty());
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
            });
            Thread receptor = new Thread(() -> {
                try {
                    for (Optional<String> m = cola.take(); m.isPresent(); m = cola.take()) {
                        recibidos.add(m.get());
                    }
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
            });
            emisor.start();
            receptor.start();
            emisor.join();
            receptor.join();
            return recibidos;
        }
    }

    public class Almacen {
        private final int id;
        private int piezas;

        public Almacen(int id, int piezas) {
            this.id = id;
            this.piezas = piezas;
        }

        public synchronized int piezas() {
            return piezas;
        }

        public static void mover(Almacen de, Almacen a, int n) {
            Almacen primero = de.id < a.id ? de : a;
            Almacen segundo = primero == de ? a : de;
            synchronized (primero) {
                synchronized (segundo) {
                    if (de.piezas < n) {
                        throw new IllegalStateException("no hay bastantes piezas");
                    }
                    de.piezas -= n;
                    a.piezas += n;
                }
            }
        }

        /** Ya hecho: dos hilos se pasan piezas a la vez, en sentidos contrarios. */

            Almacen sur = new Almacen(2, 1000);

                for (int i = 0; i < veces; i++) {
                    mover(norte, sur, 1);
                }
            });
            Thread vuelta = new Thread(() -> {
                for (int i = 0; i < veces; i++) {
                    mover(sur, norte, 2);
                }
            });

            vuelta.start();
            ida.join();
            vuelta.join();
