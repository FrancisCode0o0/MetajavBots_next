solucion:
    import java.util.ArrayList;
    import java.util.Collections;
    import java.util.HashMap;
    import java.util.List;
    import java.util.Map;
    import java.util.function.Consumer;

    public class Radio {
        private final Map<String, List<Consumer<String>>> oyentes = new HashMap<>();

        public Radio escuchar(String tipo, Consumer<String> oyente) {
            oyentes.computeIfAbsent(tipo, t -> new ArrayList<>()).add(oyente);
            return this;
        }

        public int emitir(String tipo, String mensaje) {
            List<Consumer<String>> lista = oyentes.getOrDefault(tipo, Collections.emptyList());
            lista.forEach(oyente -> oyente.accept(mensaje));
            return lista.size();
        }
    }
