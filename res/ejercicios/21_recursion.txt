        }
    }
solucion:

    import java.util.Map;

    public class Fibonacci {
        private static final Map<Integer, Long> memoria = new HashMap<>();

        public static long fib(int n) {
            if (n < 0) {
                throw new IllegalArgumentException("n negativo: " + n);
            }
            if (n < 2) {
                return n;
            }
            Long guardado = memoria.get(n);
            if (guardado != null) {

        }

enunciado.en: The array is sorted in ascending order. Return the position of objetivo between desde (inclusive) and hasta (exclusive), or -1 if it isn't there. Make it recursive with binary search: look at the middle and carry on only in the half where it can be.

pista.en: If desde >= hasta, it isn't there. int medio = (desde + hasta) >>> 1; if ordenado[medio] < objetivo, carry on in (medio + 1, hasta); if it is larger, in (desde, medio).

        return -1;

        return medio;
    }

    }

pista.es: Caso base: con 0 o 1 elementos, devuelve datos.clone(). Las mitades, con Arrays.copyOfRange. Para mezclar, tres índices: uno por mitad y otro para el resultado.

firma: static int[] ordenar(int[] datos)
plantilla:
    return datos;
solucion:
    if (datos.length <= 1) {
        return datos.clone();
    }
    int mitad = datos.length / 2;
    int[] izquierda = ordenar(Arrays.copyOfRange(datos, 0, mitad));
    int[] derecha = ordenar(Arrays.copyOfRange(datos, mitad, datos.length));
    int[] resultado = new int[datos.length];
    int i = 0, j = 0, k = 0;
    while (i < izquierda.length && j < derecha.length) {
        resultado[k++] = izquierda[i] <= derecha[j] ? izquierda[i++] : derecha[j++];
    }
    while (i < izquierda.length) {
        resultado[k++] = izquierda[i++];
    }
    while (j < derecha.length) {
        resultado[k++] = derecha[j++];
    }
    return resultado;

enunciado.en: The map has a robot 'R', an enemy 'E', floor '.' and walls '#' (all rows are the same length). Return the smallest number of steps (up, down, left or right) from R to E, or -1 if it can't be done.
pista.es: Búsqueda en anchura (BFS): una cola ArrayDeque<int[]> con R a distancia 0; saca una casilla, y a cada vecina libre que no hayas visto ponle distancia + 1 y métela en la cola. La primera vez que sacas E, esa es la distancia mínima.
pista.de: Breitensuche (BFS): eine Queue ArrayDeque<int[]> mit R bei Entfernung 0; nimm ein Feld heraus und gib jedem freien, noch nicht besuchten Nachbarn Entfernung + 1 und leg ihn in die Queue. Beim ersten Herausnehmen von E hast du die kürzeste Entfernung.
pista.en: Breadth-first search (BFS): an ArrayDeque<int[]> queue with R at distance 0; take a cell out, and give every free neighbour you haven't seen distance + 1 and add it to the queue. The first time you take E out, that is the shortest distance.
firma: static int pasosHastaEnemigo(String[] mapa)
plantilla:
    return -1;
solucion:
    int filas = mapa.length;
    int columnas = filas == 0 ? 0 : mapa[0].length();
    int[][] distancia = new int[filas][columnas];
    Deque<int[]> cola = new ArrayDeque<>();
    for (int f = 0; f < filas; f++) {
        Arrays.fill(distancia[f], -1);
        int c = mapa[f].indexOf('R');
        if (c >= 0) {
            distancia[f][c] = 0;
            cola.add(new int[]{f, c});
        }
    }
    int[][] direcciones = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
    while (!cola.isEmpty()) {
        int[] actual = cola.poll();
        int pasos = distancia[actual[0]][actual[1]];
        if (mapa[actual[0]].charAt(actual[1]) == 'E') {
            return pasos;
        }
        for (int[] d : direcciones) {
            int f = actual[0] + d[0];
            int c = actual[1] + d[1];
            if (f >= 0 && f < filas && c >= 0 && c < columnas && mapa[f].charAt(c) != '#' && distancia[f][c] < 0) {
                distancia[f][c] = pasos + 1;
                cola.add(new int[]{f, c});
            }
        }
    }
    return -1;

Return how many ways the remaining ones can be placed. The first call is colocar(n, 0, new int[n]).
pista.es: Si fila == n, ya están todas: 1 forma. Si no, prueba cada columna c: si no choca con ninguna fila anterior f (columnas[f] != c y Math.abs(columnas[f] - c) != fila - f), apúntala en columnas[fila] y suma colocar(n, fila + 1, columnas).

pista.en: If fila == n, they are all placed: 1 way. Otherwise try each column c: if it doesn't clash with any earlier row f (columnas[f] != c and Math.abs(columnas[f] - c) != fila - f), record it in columnas[fila] and add colocar(n, fila + 1, columnas).
firma: static int colocar(int n, int fila, int[] columnas)
plantilla:
    return 0;
solucion:
    if (fila == n) {
        return 1;
    }
    int formas = 0;
    for (int c = 0; c < n; c++) {
        boolean libre = true;
        for (int f = 0; f < fila && libre; f++) {
            libre = columnas[f] != c && Math.abs(columnas[f] - c) != fila - f;
        }
        if (libre) {
            columnas[fila] = c;
            formas += colocar(n, fila + 1, columnas);
        }
    }
    return formas;
