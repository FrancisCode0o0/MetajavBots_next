    class Pieza {
        private final String nombre;
        private final int peso;

        Pieza(String nombre, int peso) {
            this.nombre = nombre;

        public boolean equals(Object otro) {
            if (this == otro) {
                return true;

                return false;
            }
            Pieza p = (Pieza) otro;

    public class PilaConMinimo extends Pila {
        private final Pila minimos = new Pila();

        @Override
        PilaConMinimo apilar(int valor) {
            if (minimos.estaVacia() || valor <= minimos.cima()) {
                minimos.apilar(valor);
            }
            super.apilar(valor);
            return this;
        }

        @Override
        int desapilar() {
            int valor = super.desapilar();
            if (valor == minimos.cima()) {
                minimos.desapilar();
            }
            return valor;
        }

        int minimo() {
            if (estaVacia()) {
                throw new IllegalStateException("pila vacia");
            }
            return minimos.cima();
        }
