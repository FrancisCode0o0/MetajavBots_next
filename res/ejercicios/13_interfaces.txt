        boolean detecta(int distancia);
    }

    class Radar implements Sensor {
        private final int alcance;

        Radar(int alcance) {
            this.alcance = alcance;
        }

        @Override
        public boolean detecta(int distancia) {
            return distancia <= alcance;
        }
    }

    class Camara implements Sensor {
        @Override
        public boolean detecta(int distancia) {
            return distancia >= 2 && distancia <= 15;
        }
    }

        private Sensor sensor;

        public Robot(Sensor sensor) {
            this.sensor = sensor;
        }

            sensor = nuevo;
            return this;

            int cuantos = 0;
            for (int d : distancias) {
                if (sensor.detecta(d)) {
                    cuantos++;
                }
            }
            return cuantos;
        }

    public class Robot implements Comparable<Robot> {
        private final String nombre;
        private final int potencia;

        public Robot(String nombre, int potencia) {
            this.nombre = nombre;
            this.potencia = potencia;
        }

        public String getNombre() {
            return nombre;
        }

        @Override
        public String toString() {
            return nombre;
        }

        @Override
        public int compareTo(Robot otro) {
            int porPotencia = Integer.compare(potencia, otro.potencia);

    public interface Arma {
        int dano();

        default int danoCritico() {
            return dano() * 2;
        }

        static Arma de(int dano) {
            if (dano < 0) {
                throw new IllegalArgumentException("dano negativo: " + dano);
            }
            return () -> dano;
        }

        static Arma mejor(Arma a, Arma b) {
            return b.dano() > a.dano() ? b : a;
        }
    }

        public int dano() {
            return 6;
        }
    }

    import java.util.Iterator;
    import java.util.NoSuchElementException;

    public class CuentaAtras implements Iterable<Integer> {
        private final int desde;

        public CuentaAtras(int desde) {
            if (desde < 0) {
                throw new IllegalArgumentException("cuenta negativa: " + desde);
            }
            this.desde = desde;
        }

        @Override
        public Iterator<Integer> iterator() {
            return new Paso(desde);
        }

        private static class Paso implements Iterator<Integer> {
            private int actual;

            Paso(int desde) {
                this.actual = desde;
            }

            @Override
            public boolean hasNext() {
                return actual > 0;
            }

            @Override
            public Integer next() {
                if (!hasNext()) {
                    throw new NoSuchElementException();
                }
                return actual--;
            }
        }
    }
