                .thenComparingInt(Version::menor)
                .thenComparingInt(Version::parche);

                throw new IllegalArgumentException("versión con números negativos");

                throw new IllegalArgumentException("versión no válida: " + texto);

                return new Version(Integer.parseInt(partes[0]), Integer.parseInt(partes[1]), Integer.parseInt(partes[2]));

                throw new IllegalArgumentException("versión no válida: " + texto, e);
            }
        }

        @Override
        public int compareTo(Version otra) {
            return ORDEN.compare(this, otra);

    import java.util.ArrayDeque;
    import java.util.ArrayList;
    import java.util.Deque;
    import java.util.List;

    interface Orden {
        void hacer(List<String> piezas);

        void deshacer(List<String> piezas);
    }

    final class Poner implements Orden {
        private final String pieza;

        Poner(String pieza) {
            this.pieza = pieza;
        }

        public void hacer(List<String> piezas) {
            piezas.add(pieza);
        }

        public void deshacer(List<String> piezas) {
            piezas.remove(piezas.size() - 1);
        }
    }

    final class Cambiar implements Orden {
        private final int posicion;
        private final String nueva;
        private String anterior;

        Cambiar(int posicion, String nueva) {
            this.posicion = posicion;
            this.nueva = nueva;
        }

        public void hacer(List<String> piezas) {
            anterior = piezas.set(posicion, nueva);
        }

        public void deshacer(List<String> piezas) {
            piezas.set(posicion, anterior);
        }
    }

    public class Taller {
        private final List<String> piezas = new ArrayList<>();
        private final Deque<Orden> historial = new ArrayDeque<>();

        public Taller hacer(Orden orden) {
            orden.hacer(piezas);
            historial.push(orden);
            return this;
        }

        public Taller deshacer() {
            if (!historial.isEmpty()) {
                historial.pop().deshacer(piezas);
            }
            return this;
        }

        public List<String> piezas() {
            return List.copyOf(piezas);
        }

    public class Enfriamiento {
