enunciado.en: Remove from the list golpes (the same one, not a copy) the hits below minimo and return it. The template removes inside a for-each and Java throws ConcurrentModificationException: you can't change a list while looping over it like that. Fix it with an Iterator.

pista.en: Iterator<Integer> it = golpes.iterator(); while (it.hasNext()) { if (it.next() < minimo) { it.remove(); } } — or, since Java 8, golpes.removeIf(g -> g < minimo).

        }

    }
solucion:
    import java.util.ArrayDeque;
    import java.util.Deque;

    public class Taller {
        private static final String NADIE = "nadie";

        private final Deque<String> cola = new ArrayDeque<>();

        public Taller llega(String robot) {
            cola.addLast(robot);
            return this;
        }

        public Taller urgente(String robot) {
            cola.addFirst(robot);
            return this;
        }

        public String atender() {
            String primero = cola.pollFirst();
            return primero == null ? NADIE : primero;
        }

        public String siguiente() {
            String primero = cola.peekFirst();
            return primero == null ? NADIE : primero;
        }

        public int enEspera() {
            return cola.size();
        }
    }

pista.en: Split with split(":"). If the level isn't in the map yet, put a new list in; then add the name to that level's list. computeIfAbsent does both in one go.
firma: static Map<Integer, List<String>> porNivel(List<String> fichas)
plantilla:
    return new HashMap<>();
solucion:
    Map<Integer, List<String>> grupos = new TreeMap<>();
    for (String ficha : fichas) {
        String[] partes = ficha.split(":");
        int nivel = Integer.parseInt(partes[1]);
        List<String> nombres = grupos.get(nivel);
        if (nombres == null) {
            nombres = new ArrayList<>();
            grupos.put(nivel, nombres);
        }
        nombres.add(partes[0]);
    }
    return grupos;

        public Inventario meter(String pieza, int cantidad) {
            if (cantidad <= 0) {
                throw new IllegalArgumentException("cantidad no positiva: " + cantidad);
            }
            piezas.put(pieza, cantidad(pieza) + cantidad);
            return this;
        }

        public boolean sacar(String pieza, int cantidad) {
            int hay = cantidad(pieza);
            if (cantidad > hay) {
                return false;
            }
            if (cantidad == hay) {
                piezas.remove(pieza);
            } else {
                piezas.put(pieza, hay - cantidad);
            }
            return true;
        }

        public int cantidad(String pieza) {
            return piezas.getOrDefault(pieza, 0);
        }

        public int total() {
            int total = 0;
            for (int n : piezas.values()) {
                total += n;
            }
            return total;
        }

        public Set<String> piezas() {
            return Collections.unmodifiableSet(new TreeSet<>(piezas.keySet()));
        }
    }
