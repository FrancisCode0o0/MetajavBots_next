            return 0;
        }

        public static boolean esCapicua(int n) {
            return false;
        }
    }
solucion:
    public class Cifras {

        public static int invertir(int n) {
            int resultado = 0;
            while (n > 0) {
                resultado = resultado * 10 + n % 10;
                n /= 10;
            }
            return resultado;
        }

        public static boolean esCapicua(int n) {
            return invertir(n) == n;
        }
    }

        public static int danioBase(int nivel) {
            return 5 + 3 * nivel;
        }

        public static double multiplicador(String arma) {

                return 1.5;
