    import java.util.Objects;

    public class Robot {
        private final String nombre;
        private final int vida;

        public Robot(String nombre, int vida) {
            Objects.requireNonNull(nombre, "falta el nombre");
            if (nombre.trim().isEmpty()) {
                throw new IllegalArgumentException("nombre vacio");
            }
            if (vida < 1 || vida > 100) {
                throw new IllegalArgumentException("vida fuera de rango: " + vida);
            }
            this.nombre = nombre;
            this.vida = vida;
        }

        public String getNombre() {
            return nombre;
        }

        public int getVida() {
            return vida;
        }
    }

If the argument is wrong (loading 0 or less), IllegalArgumentException; if what's wrong is the turret's state (firing while empty, or loading more than 10 bullets in total), IllegalStateException.

plantilla:

        // cargar, disparar y estaCargada
    }
solucion:
    public class Torreta {
        private static final int CAPACIDAD = 10;

        private int balas;

        public Torreta cargar(int balas) {
            if (balas <= 0) {
                throw new IllegalArgumentException("hay que cargar al menos una bala");
            }
            if (this.balas + balas > CAPACIDAD) {
                throw new IllegalStateException("no caben mas de " + CAPACIDAD + " balas");
            }
            this.balas += balas;
            return this;
        }

        public int disparar() {
            if (balas == 0) {
                throw new IllegalStateException("torreta vacia");
            }
            balas--;
            return balas;
        }

        public boolean estaCargada() {
            return balas > 0;
        }
    }

    new Torreta().cargar(0) => !IllegalArgumentException

    class SobrecalentamientoException extends Exception {
        private final int temperatura;

        SobrecalentamientoException(int temperatura) {
            super("temperatura " + temperatura);
            this.temperatura = temperatura;
        }

        int getTemperatura() {
            return temperatura;
        }

    public class Reactor {
        private static final int LIMITE = 100;

        private int temperatura = 20;

        public int calentar(int grados) throws SobrecalentamientoException {
            int nueva = temperatura + grados;
            if (nueva > LIMITE) {
                throw new SobrecalentamientoException(nueva);

            temperatura = nueva;
            return temperatura;

                return "ok " + new Reactor().calentar(grados);

                return "peligro " + e.getTemperatura();
            }
        }

        Canal(StringBuilder registro) {

        }

            }

        }

        public void close() {
            registro.append("cierra;");
        }
    }

        public static String transmitir(String... mensajes) {

                }

            }

        }

    class SinConexionException extends Exception {
        SinConexionException(int intentos, Throwable causa) {
            super("sin conexion tras " + intentos + " intentos", causa);
        }
    }

    public class Conector {
        public static int conectar(Enlace enlace, int maxIntentos) throws SinConexionException {
            if (maxIntentos < 1) {
                throw new IllegalArgumentException("hace falta al menos un intento");
            }
            IllegalStateException ultima = null;
            for (int intento = 1; intento <= maxIntentos; intento++) {
                try {
                    enlace.conectar();
                    return intento;
                } catch (IllegalStateException e) {
                    ultima = e;
                }
            }
            throw new SinConexionException(maxIntentos, ultima);
        }
    }
