        private final String nombre;
        private final int vida;

        protected Robot(String nombre, int vida) {
            this.nombre = nombre;
            this.vida = vida;
        }

        public int getVida() {
            return vida;
        }

        public abstract int ataque();

        public String ficha() {
            return nombre + " (vida " + vida + ", ataque " + ataque() + ")";
        }
    }

    class Tanque extends Robot {
        Tanque(String nombre) {
            super(nombre, 100);
        }

        @Override
        public int ataque() {
            return 8;
        }
    }

        Explorador(String nombre) {
            super(nombre, 40);
        }

        @Override
        public int ataque() {

        }

pista.en: In areaTotal you don't care which zone each one is: z.area() calls each object's version. The circle's area is Math.PI * radio * radio.
plantilla:
    public abstract class Zona {
        // area() abstracto y areaTotal
    }

    // Circulo y Rectangulo
solucion:
    public abstract class Zona {
        public abstract double area();

        public static double areaTotal(Zona[] zonas) {
            double total = 0;
            for (Zona z : zonas) {
                total += z.area();
            }
            return total;
        }
    }

    class Circulo extends Zona {
        private final double radio;

        Circulo(double radio) {
            this.radio = radio;
        }

        @Override
        public double area() {
            return Math.PI * radio * radio;
        }
    }

    class Rectangulo extends Zona {
        private final double ancho;
        private final double alto;

        Rectangulo(double ancho, double alto) {
            this.ancho = ancho;
            this.alto = alto;
        }

        @Override
        public double area() {
            return ancho * alto;
        }
    }

    new Circulo(1).area() => Math.PI
    Zona.areaTotal(new Zona[]{new Rectangulo(2, 3), new Circulo(2)}) => 6 + 4 * Math.PI
    Zona.areaTotal(new Zona[]{new Rectangulo(0, 5)}) => 0.0
    Zona.areaTotal(new Zona[0]) => 0.0

enunciado.en: Write the abstract class Pieza with Pieza(String nombre), the abstract method String estado(), String informe() ("<nombre>: <estado>") and static String revisar(Pieza[] piezas), which joins the reports with "; ". Then Rueda(int desgaste) (name "rueda"; "gastada" if the wear is over 50, otherwise "bien") and Laser(int disparos) (name "laser"; "recargar" with no shots, otherwise "listo").
pista.es: Rueda(int desgaste) { super("rueda"); this.desgaste = desgaste; }. En revisar, un StringBuilder y pon "; " delante de todos menos del primero.
pista.de: Rueda(int desgaste) { super("rueda"); this.desgaste = desgaste; }. In revisar ein StringBuilder, und "; " vor alle außer dem ersten.
pista.en: Rueda(int desgaste) { super("rueda"); this.desgaste = desgaste; }. In revisar, use a StringBuilder and put "; " before all but the first.
plantilla:
    public abstract class Pieza {
        // nombre, constructor, estado() abstracto, informe() y revisar
    }

    // Rueda y Laser
solucion:
    public abstract class Pieza {
        private final String nombre;

        protected Pieza(String nombre) {
            this.nombre = nombre;
        }

        public abstract String estado();

        public String informe() {
            return nombre + ": " + estado();
        }

        public static String revisar(Pieza[] piezas) {
            StringBuilder texto = new StringBuilder();
            for (Pieza p : piezas) {
                if (texto.length() > 0) {
                    texto.append("; ");
                }
                texto.append(p.informe());
            }
            return texto.toString();
        }
    }

    class Rueda extends Pieza {
        private final int desgaste;

        Rueda(int desgaste) {
            super("rueda");
            this.desgaste = desgaste;
        }

        @Override
        public String estado() {
            return desgaste > 50 ? "gastada" : "bien";
        }
    }

    class Laser extends Pieza {
        private final int disparos;

        Laser(int disparos) {
            super("laser");
            this.disparos = disparos;
        }

        @Override
        public String estado() {
            return disparos == 0 ? "recargar" : "listo";
        }
    }

plantilla:
    public abstract class Mision {
        // recompensa() abstracto, esDificil() y total
    }

    // Escolta, Asalto y Rescate
solucion:
    public abstract class Mision {
        public abstract int recompensa();

        public boolean esDificil() {
            return recompensa() > 100;
        }

        public static int total(Mision[] misiones) {
            int total = 0;
            for (Mision m : misiones) {
                total += m.recompensa();
            }
            return total;
        }
    }

    class Escolta extends Mision {
        private final int robots;

        Escolta(int robots) {
            this.robots = robots;
        }

        @Override
        public int recompensa() {
            return 30 * robots;
        }
    }

    class Asalto extends Mision {
        private final int torres;

        Asalto(int torres) {
            this.torres = torres;
        }

        @Override
        public int recompensa() {
            return 50 * torres + 20;
        }
    }

    class Rescate extends Mision {
        @Override
        public int recompensa() {
            return 120;
        }
    }

            this.nombre = nombre;
        }

        }

    class Arma extends Pieza {

            int cuantas = 0;
            for (int d : distancias) {
                if (detecta(d)) {
                    cuantas++;
                }
            }
            return cuantas;
        }

        static Sensor deAlcance(int alcance) {
            return new Sensor() {
                @Override
                boolean detecta(int distancia) {
                    return distancia <= alcance;
                }
            };
        }

        static Sensor entre(int minimo, int maximo) {
            return new Sensor() {
                @Override
                boolean detecta(int distancia) {
                    return distancia >= minimo && distancia <= maximo;
                }
            };
        }

        static Sensor ciego() {
            return new Sensor() {
                @Override
                boolean detecta(int distancia) {
                    return false;
                }
            };
        }
    }

plantilla:

        abstract int aplicar(int dano);
    }

    // Duplicar, Reducir, Tope y Combinado
solucion:
    public abstract class Efecto {
        abstract int aplicar(int dano);
    }

    class Duplicar extends Efecto {
        @Override
        int aplicar(int dano) {
            return dano * 2;
        }
    }

    class Reducir extends Efecto {
        private final int cantidad;

        Reducir(int cantidad) {
            this.cantidad = cantidad;
        }

        @Override
        int aplicar(int dano) {
            return Math.max(0, dano - cantidad);
        }
    }

    class Tope extends Efecto {
        private final int maximo;

        Tope(int maximo) {
            this.maximo = maximo;
        }

        @Override
        int aplicar(int dano) {
            return Math.min(dano, maximo);
        }
    }

    class Combinado extends Efecto {
        private final Efecto[] efectos;

        Combinado(Efecto... efectos) {
            this.efectos = efectos.clone();
        }

        @Override
        int aplicar(int dano) {
            for (Efecto e : efectos) {
                dano = e.aplicar(dano);
            }
            return dano;
        }
    }

    public abstract class Estado {
        abstract String nombre();

        abstract Estado siguiente(String evento);

        static Estado inicial() {
            return new Patrullando();
        }

        Estado procesar(String... eventos) {
            Estado actual = this;
            for (String evento : eventos) {
                actual = actual.siguiente(evento);
            }
            return actual;
        }
    }

    class Patrullando extends Estado {
        @Override
        String nombre() {
            return "patrullando";
        }

        @Override
        Estado siguiente(String evento) {
            switch (evento) {
                case "enemigo":
                    return new Atacando();
                case "herido":
                    return new Huyendo();
                default:
                    return this;
            }
        }
    }

    class Atacando extends Estado {
        @Override
        String nombre() {
            return "atacando";
        }

        @Override
        Estado siguiente(String evento) {
            switch (evento) {
                case "herido":
                    return new Huyendo();
                case "victoria":
                    return new Patrullando();
                default:
                    return this;
            }
        }
    }

    class Huyendo extends Estado {
        @Override
        String nombre() {
            return "huyendo";
        }

        @Override
        Estado siguiente(String evento) {
            return evento.equals("reparado") ? new Patrullando() : this;
        }
    }
