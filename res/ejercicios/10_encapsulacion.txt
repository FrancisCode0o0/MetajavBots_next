    public class BarraDeVida {
        private final int maximo;
        private int vida;

        public BarraDeVida(int maximo) {
            if (maximo <= 0) {
                throw new IllegalArgumentException("el maximo tiene que ser positivo");
            }
            this.maximo = maximo;
            this.vida = maximo;
        }

        public int getVida() {
            return vida;
        }

        public int danar(int dano) {
            if (dano < 0) {
                throw new IllegalArgumentException("dano negativo");
            }
            vida = Math.max(0, vida - dano);
            return vida;
        }

        public int curar(int puntos) {
            if (puntos < 0) {
                throw new IllegalArgumentException("curacion negativa");
            }
            vida = Math.min(maximo, vida + puntos);
            return vida;
        }

        public boolean estaVacia() {
            return vida == 0;
        }
    }

    public class Monedero {
        private int creditos;

        public Monedero(int creditos) {
            if (creditos < 0) {
                throw new IllegalArgumentException("creditos negativos");
            }
            this.creditos = creditos;
        }

        public int getCreditos() {
            return creditos;
        }

        public boolean pagar(int precio) {
            if (precio < 0) {
                throw new IllegalArgumentException("precio negativo");
            }
            if (precio > creditos) {
                return false;
            }
            creditos -= precio;
            return true;
        }

            if (cantidad <= 0) {
                throw new IllegalArgumentException("cantidad no positiva");
            }
            creditos += cantidad;
        }

    public class Cronometro {
        private final int totalSegundos;

        public Cronometro(int minutos, int segundos) {
            if (minutos < 0 || segundos < 0) {
                throw new IllegalArgumentException("tiempo negativo");
            }
            this.totalSegundos = minutos * 60 + segundos;
        }

        public int getMinutos() {
            return totalSegundos / 60;
        }

        public int getSegundos() {
            return totalSegundos % 60;
        }

        public Cronometro mas(int segundos) {
            return new Cronometro(0, totalSegundos + segundos);
        }

        @Override
        public String toString() {
            return String.format("%02d:%02d", getMinutos(), getSegundos());
        }
    }

    public class Reactor {
        private static final double CERO_ABSOLUTO = -273.15;

        private final double celsius;

        private Reactor(double celsius) {
            if (celsius < CERO_ABSOLUTO) {
                throw new IllegalArgumentException("por debajo del cero absoluto");
            }
            this.celsius = celsius;
        }

        public static Reactor enCelsius(double grados) {
            return new Reactor(grados);
        }

        public static Reactor enFahrenheit(double grados) {
            return new Reactor((grados - 32) * 5 / 9);

        private final String clave;
        private int intentosRestantes = 3;

        public Candado(String clave) {
            this.clave = clave;
        }

        public boolean abrir(String intento) {
            if (estaBloqueado()) {
                return false;
            }
            if (clave.equals(intento)) {
                return true;

    public class NumeroDeSerie {
        private static final int MAXIMO = 9999;

        private final String serie;
        private final int numero;

        private NumeroDeSerie(String serie, int numero) {
            this.serie = serie;
            this.numero = numero;
        }

        public static NumeroDeSerie de(String texto) {
            if (texto == null || !texto.matches("[A-Z]{2,}-[0-9]{4}")) {
                throw new IllegalArgumentException("numero de serie mal escrito: " + texto);
            }
            int guion = texto.indexOf('-');
            return new NumeroDeSerie(texto.substring(0, guion), Integer.parseInt(texto.substring(guion + 1)));
        }

        public String getSerie() {
            return serie;
        }

        public int getNumero() {
            return numero;
        }

        public NumeroDeSerie siguiente() {
            if (numero == MAXIMO) {
                throw new IllegalStateException("la serie " + serie + " se ha acabado");
            }
            return new NumeroDeSerie(serie, numero + 1);
        }

        @Override
        public String toString() {
            return String.format("%s-%04d", serie, numero);
        }
    }

    public class Robot {
        private final String nombre;
        private final int blindaje;
        private final int velocidad;
        private final String arma;

        private Robot(Builder b) {
            this.nombre = b.nombre;
            this.blindaje = b.blindaje;
            this.velocidad = b.velocidad;
            this.arma = b.arma;
        }

        public int getVelocidad() {
            return velocidad;
        }

        @Override
        public String toString() {
            return nombre + "[blindaje=" + blindaje + ", velocidad=" + velocidad + ", arma=" + arma + "]";
        }

        public static class Builder {
            private final String nombre;
            private int blindaje = 0;
            private int velocidad = 5;
            private String arma = "puno";

            public Builder(String nombre) {
                this.nombre = nombre;
            }

            public Builder blindaje(int blindaje) {
                this.blindaje = blindaje;
                return this;
            }

            public Builder velocidad(int velocidad) {
                this.velocidad = velocidad;
                return this;
            }

            public Builder arma(String arma) {
                this.arma = arma;
                return this;
            }

            public Robot build() {
                if (blindaje < 0 || blindaje > 10) {
                    throw new IllegalStateException("blindaje fuera de rango: " + blindaje);
                }
                if (velocidad < 1 || velocidad > 10) {
                    throw new IllegalStateException("velocidad fuera de rango: " + velocidad);
                }
                return new Robot(this);
            }
        }
