    import java.util.Arrays;
    import java.util.Comparator;
    import java.util.Optional;

    public enum Arma {
        LASER(12, 600), BALA(8, 400), MISIL(25, 300), LLAMA(30, 100);

        private final int dano;
        private final int alcance;

        Arma(int dano, int alcance) {
            this.dano = dano;
            this.alcance = alcance;
        }

        public int dano() {
            return dano;
        }

        public int alcance() {
            return alcance;
        }

        public boolean alcanza(int distancia) {
            return distancia <= alcance;
        }

        public static Optional<Arma> mejorPara(int distancia) {
            return Arrays.stream(values())
                    .filter(a -> a.alcanza(distancia))
                    .max(Comparator.comparingInt(Arma::dano));
        }
    }

    public sealed interface Formula permits Num, Suma, Producto, Neg {

        static int evaluar(Formula f) {
            return switch (f) {
                case Num(var valor) -> valor;
                case Suma(var a, var b) -> evaluar(a) + evaluar(b);
                case Producto(var a, var b) -> evaluar(a) * evaluar(b);
                case Neg(var a) -> -evaluar(a);
            };
        }

        static String mostrar(Formula f) {
            return switch (f) {
                case Num(var valor) -> String.valueOf(valor);
                case Suma(var a, var b) -> "(" + mostrar(a) + " + " + mostrar(b) + ")";
                case Producto(var a, var b) -> "(" + mostrar(a) + " * " + mostrar(b) + ")";
                case Neg(var a) -> "-" + mostrar(a);
            };
        }

        static Formula simplificar(Formula f) {
            return switch (f) {
                case Num n -> n;
                case Neg(var a) -> switch (simplificar(a)) {
                    case Neg(var dentro) -> dentro;
                    case Formula otra -> new Neg(otra);
                };
                case Suma(var a, var b) -> switch (new Suma(simplificar(a), simplificar(b))) {
                    case Suma(Num(var c), var x) when c == 0 -> x;
                    case Suma(var x, Num(var c)) when c == 0 -> x;
                    case Suma s -> s;
                };
                case Producto(var a, var b) -> switch (new Producto(simplificar(a), simplificar(b))) {
                    case Producto(Num(var c), _) when c == 0 -> new Num(0);
                    case Producto(_, Num(var c)) when c == 0 -> new Num(0);
                    case Producto(Num(var c), var x) when c == 1 -> x;
                    case Producto(var x, Num(var c)) when c == 1 -> x;
                    case Producto p -> p;
                };
            };
        }
    }
